class Solution {
public:
    void twoSum(vector<int>& arr, int sum, int index, set<vector<int>> &st){
        int left =0, right = arr.size() -1;

        while(left < index && right > index){
            if(left == index) left++;
            else if(right == index) right--;

            if(arr[left] + arr[right] == sum){
                // set<int> idxs{left, right, index};
                // vector<int> sortedIndex(idxs.begin(), idxs.end());
                // st.insert({arr[sortedIndex[0]], arr[sortedIndex[1]], arr[sortedIndex[2]]});
                st.insert({arr[left], arr[index], arr[right]});
                left++;
            }
            else if(arr[left] + arr[right] < sum) left++;
            else right--;
        }
    }
    vector<vector<int>> threeSum(vector<int>& arr) {
        set<vector<int>> st;
        sort(arr.begin(), arr.end());
        int totalSum = 0;
        for(int i=0;i<arr.size();i++){
            int sum = totalSum - arr[i];
            twoSum(arr, sum, i, st);
        }
        return {st.begin(), st.end()};
    }
};

class Solution:
    def threeSum1(self, nums: List[int]) -> List[List[int]]:
        result = []
        nums.sort()
        print(nums)
        for i in range(len(nums)):
            if i >0 and nums[i] == nums[i-1]:
                continue
            left = i+1
            right = len(nums)-1

            while left < right:
                total =  nums[i]+ nums[left] + nums[right ]
                if total ==0:
                    result.append([nums[i],nums[left], nums[right]])
                    left +=1
                    right -=1

                    # avoid duplicates
                    while left < right and nums[left] == nums[left-1]:
                        left +=1

                elif total < 0:
                    left +=1
                else:
                    right -=1

        return result

    def threeSum(self, nums: List[int]) -> List[List[int]]:
        '''
        1. sort the list
        2. run outer loop for each element till last-2
        3. avoid duplicate processing for outer 
        4. run inner loop with while loop - i+1 .. and add this element to the seen complement list
        5. find complement in the seen list
        6. if found save triplet and skip duplicates

        nlog n+ n^2
        '''
        result= []
        #1. sort the list first
        nums.sort()
        #print(nums)

        # for each elemetn in the outer loop
        for i in range(len(nums)-1):

            # avoid duplicate in first element
            if i >0 and nums[i-1] == nums[i]:
                continue
            j = i+1
            seen = set()

            # for each element after the outer loop element
            while j< len(nums):

                # find complement of outer + inner loop number
                complement = -(nums[i] + nums[j])


                # if complement found
                if complement in seen:

                    # got the triplet
                    result.append([nums[i], nums[j], complement])

                    # avoid duplicate in second element
                    while j + 1 < len(nums) and nums[j] == nums[j+1]:
                        j+=1

                # add each inner number which will become the complement
                seen.add(nums[j])
                j+=1
                
        return result
