class Solution {
public:
    vector<vector<int>> updateMatrix(vector<vector<int>>& grid) {
        int n = grid.size();
        int m = grid[0].size();

        queue<pair<int,int>> qu;
        vector<vector<int>> visited(n, vector<int>(m, -1));
        vector<vector<int>> ans(n, vector<int>(m, INT_MAX));

        for(int i=0;i<n;i++){
            for(int j=0;j<m;j++){
                if(grid[i][j] == 0){
                    visited[i][j]  = 0;
                    qu.push({i,j});
                }
            }
        }

        vector<int> dx = {0,1,0,-1};
        vector<int> dy = {1,0,-1,0};
        int dis = 0;
        while(!qu.empty()){
            int sz = qu.size();

            while(sz--){
                int x = qu.front().first;
                int y = qu.front().second;
                qu.pop();
                for(int i=0;i<dx.size();i++){
                    int r = x+ dx[i];
                    int c = y+ dy[i];

                    if(r>=0 && r<n && c>=0 && c<m &&
                        visited[r][c] == -1 && grid[r][c] == 1){
                            cout<<r<<"|"<<c<<"::";
                            visited[r][c] = dis + 1;
                            qu.push({r,c}); // don't forget this step
                        }
                }
            }
            dis++;

        }

        return visited;
    }
};
